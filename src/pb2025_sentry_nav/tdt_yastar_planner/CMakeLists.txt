cmake_minimum_required(VERSION 3.8)
project(tdt_yastar_planner)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(ament_cmake REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(Eigen3 REQUIRED)

add_library(tdt_yastar_planner SHARED src/yastar_planner.cpp src/tdt_yastar.cpp)
target_include_directories(tdt_yastar_planner PUBLIC include)
target_link_libraries(tdt_yastar_planner Eigen3::Eigen)
ament_target_dependencies(tdt_yastar_planner geometry_msgs nav2_core nav2_costmap_2d nav_msgs pluginlib rclcpp rclcpp_lifecycle tf2_ros tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core tdt_yastar_planner.xml)

install(TARGETS tdt_yastar_planner DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES tdt_yastar_planner.xml TDT_LICENSE_NOTICE.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_yastar test/test_yastar.cpp)
  target_link_libraries(test_yastar tdt_yastar_planner)
  ament_target_dependencies(test_yastar nav2_costmap_2d rclcpp rclcpp_lifecycle)
endif()
ament_package()
