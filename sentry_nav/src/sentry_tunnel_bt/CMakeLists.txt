cmake_minimum_required(VERSION 3.10)
project(sentry_tunnel_bt)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Threads REQUIRED)
set(dependencies behaviortree_cpp_v3 geometry_msgs nav2_behavior_tree nav2_costmap_2d nav2_msgs nav_msgs rclcpp rclcpp_action rcl_interfaces sensor_msgs std_msgs example_interfaces tf2 tf2_geometry_msgs tf2_ros pb_omni_pid_pursuit_controller)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
add_library(sentry_tunnel_bt_node SHARED src/tunnel_passage.cpp)
target_compile_definitions(sentry_tunnel_bt_node PRIVATE BT_PLUGIN_EXPORT)
target_include_directories(sentry_tunnel_bt_node PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(sentry_tunnel_bt_node ${dependencies})
target_link_libraries(sentry_tunnel_bt_node Threads::Threads)
install(TARGETS sentry_tunnel_bt_node ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_passage_diagnostics test/test_passage_diagnostics.cpp)
  target_include_directories(test_passage_diagnostics PRIVATE include)
  ament_add_gtest(test_exit_spin_safety test/test_exit_spin_safety.cpp)
  target_include_directories(test_exit_spin_safety PRIVATE include)
  ament_target_dependencies(test_exit_spin_safety nav_msgs nav2_costmap_2d pb_omni_pid_pursuit_controller)
  ament_add_gtest(test_continuous_route test/test_continuous_route.cpp)
  target_include_directories(test_continuous_route PRIVATE include)
  ament_target_dependencies(test_continuous_route nav_msgs nav2_costmap_2d)
  ament_add_gtest(test_tunnel_geometry test/test_tunnel_geometry.cpp)
  target_include_directories(test_tunnel_geometry PRIVATE include)
  ament_target_dependencies(test_tunnel_geometry nav_msgs nav2_costmap_2d)
  ament_add_gtest(test_action_cancel test/test_action_cancel.cpp TIMEOUT 40)
  target_include_directories(test_action_cancel PRIVATE src)
  ament_target_dependencies(test_action_cancel ${dependencies})
  target_link_libraries(test_action_cancel Threads::Threads)
  ament_add_gtest(test_tunnel_route_guard test/test_tunnel_route_guard.cpp)
  ament_target_dependencies(test_tunnel_route_guard ${dependencies})
  target_compile_definitions(test_tunnel_route_guard PRIVATE
    TUNNEL_BT_PLUGIN_FILE="$<TARGET_FILE:sentry_tunnel_bt_node>"
    NAV2_PIPELINE_PLUGIN_FILE="${nav2_behavior_tree_DIR}/../../../lib/libnav2_pipeline_sequence_bt_node.so")
  add_dependencies(test_tunnel_route_guard sentry_tunnel_bt_node)
  ament_add_gtest(test_task_classification test/test_task_classification.cpp TIMEOUT 35)
  ament_target_dependencies(test_task_classification ${dependencies})
  target_compile_definitions(test_task_classification PRIVATE
    TUNNEL_BT_PLUGIN_FILE="$<TARGET_FILE:sentry_tunnel_bt_node>")
  add_dependencies(test_task_classification sentry_tunnel_bt_node)
endif()
ament_export_include_directories(include)
ament_export_libraries(sentry_tunnel_bt_node)
ament_export_dependencies(${dependencies})
ament_package()
